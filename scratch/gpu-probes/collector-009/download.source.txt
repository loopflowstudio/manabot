"""Recover a completed run's immutable evidence with bounded authenticated reads."""

from concurrent.futures import ThreadPoolExecutor
import hashlib
import json
from pathlib import Path
import tarfile
import time

import boto3
from botocore.config import Config
from cli import configure


def main() -> None:
    private = Path(__file__).parent
    root = private / "result-009"
    root.mkdir(exist_ok=True)
    attempt = json.loads((private / "lifecycle-009.json").read_text())[0]
    output = next(o for o in attempt["outputs"] if o["name"] == "evidence.tar.gz")
    size = output["size_bytes"]
    bucket = "etude-job-journals-324288363980-us-west-2"
    key = f"etude/fantasia-gpu-validation-20261014/{attempt['session']}/out/attempts/{attempt['id']}/outputs/evidence.tar.gz"
    configure()
    client = boto3.client("s3", region_name="us-west-2", config=Config(
        connect_timeout=10, read_timeout=60, retries={"max_attempts": 1}))
    started = time.monotonic()
    metadata = client.head_object(Bucket=bucket, Key=key)
    assert metadata["ContentLength"] == size

    def read_range(bounds: tuple[int, int]) -> bytes:
        start, end = bounds
        result = client.get_object(Bucket=bucket, Key=key,
                                  Range=f"bytes={start}-{end}", IfMatch=metadata["ETag"])
        with result["Body"] as body:
            payload = body.read()
        assert len(payload) == end - start + 1
        return payload

    ranges = [(start, min(start + 4 * 1024**2, size) - 1)
              for start in range(0, size, 4 * 1024**2)]
    temporary = root / "evidence.tar.gz.partial"
    digest = hashlib.sha256()
    with temporary.open("wb") as target, ThreadPoolExecutor(max_workers=4) as pool:
        for chunk in pool.map(read_range, ranges):
            digest.update(chunk)
            target.write(chunk)
    assert temporary.stat().st_size == size
    assert digest.hexdigest() == output["sha256"]
    archive = root / "evidence.tar.gz"
    temporary.replace(archive)
    with tarfile.open(archive) as bundle:
        bundle.extractall(root, filter="data")
    evidence = root / "evidence"
    manifest = json.loads((evidence / "manifest.json").read_text())
    for entry in manifest:
        path = evidence / entry["path"]
        assert path.resolve().is_relative_to(evidence.resolve())
        payload = path.read_bytes()
        assert len(payload) == entry["bytes"]
        assert hashlib.sha256(payload).hexdigest() == entry["sha256"]
    for entry in attempt["outputs"]:
        payload = (root / entry["name"]).read_bytes()
        assert len(payload) == entry["size_bytes"]
        assert hashlib.sha256(payload).hexdigest() == entry["sha256"]
    verification = {"verified_files": len(manifest), "archive_bytes": size,
                    "archive_sha256": digest.hexdigest(),
                    "authoritative_outputs_verified": len(attempt["outputs"])}
    (root / "verification.json").write_text(json.dumps(verification, indent=2) + "\n")
    receipt = {"run": attempt["run"], "bucket": bucket, "key": key,
               "previous_fetch": None,
               "method": "ordered 4MiB authenticated range GET, four readers, If-Match ETag",
               "seconds": time.monotonic() - started, **verification}
    (private / "download-recovery-009.json").write_text(json.dumps(receipt, indent=2) + "\n")
    print(json.dumps(verification))


if __name__ == "__main__":
    main()
