"""Assess frozen full-service predictions on prospective new actor counts."""
import hashlib
import json
from pathlib import Path
from statistics import median
import shutil

root=Path('.runs/gpu-validation/result-007');evidence=root/'evidence'
ledger=json.loads(Path('.runs/gpu-validation/budget.json').read_text());allocation=next(a for a in ledger['allocations'] if a['name']=='h100-validation-007')
assert allocation['status']=='terminal-deletion-confirmed'
fit_path=Path('scratch/accelerator-probes/serving-cost-calibration.json');model=json.loads(fit_path.read_text())['model']
attempts=json.loads((evidence/'attempts.json').read_text());by_name={a['name']:a for a in attempts};rows=[]
for actors in (4,16,128):
    for batch in (64,256,1024):
        name=f'serve-a{actors}-b{batch}';path=evidence/name/'result.json';attempt=by_name.get(name)
        row={'name':name,'actors':actors,'max_batch':batch,'status':attempt['status'] if attempt else 'not_attempted'}
        if path.exists() and row['status']=='completed':
            r=json.loads(path.read_text());samples=r['samples'];assert len(samples)==3
            assert r['sources'][str(fit_path)]==hashlib.sha256(fit_path.read_bytes()).hexdigest()
            assert all(s['requests']==actors*59 and s['decisions']==actors*1099 for s in samples)
            reference=samples[1]
            for s in samples:
                assert abs(sum(s['service_phase_seconds'].values())-s['service_seconds'])<1e-8
                assert 0<s['service_seconds']<=s['seconds']
            curve={}
            for size in (1,4,16,64,256,1024):
                m=json.loads((evidence/f'w64-d1-model-b{size}-s0'/'result.json').read_text());curve[size]=size/median(w['observations']/w['seconds'] for w in m['inference']['windows'])
            expected={'forward':sum(int(count)*curve[int(size)] for size,count in reference['padded_batch_counts'].items())}
            for phase,coeff in model['phases'].items():
                expected[phase]=sum(coeff['batch_seconds']+coeff['padded_row_seconds']*b['padded_rows']+coeff['request_seconds']*len(b['deliveries']) for b in reference['batches'])
            ratio=r['median_actual_to_full_service_seconds']
            row.update(predicted_seconds=r['full_service_prediction']['seconds'],original_predicted_seconds=r['prediction']['seconds'],sample_seconds=[s['seconds'] for s in samples],median_seconds=median(s['seconds'] for s in samples),actual_over_predicted=ratio,original_actual_over_predicted=r['median_actual_to_predicted_seconds'],within_20_percent=0.8<=ratio<=1.2,output_hash_differences=r['different_output_requests_vs_first'],peak_allocated_bytes=r['peak_allocated_bytes'],declared_repeat=1,service_phase_seconds=reference['service_phase_seconds'],predicted_phase_seconds_for_actual_batches=expected,actual_batches=reference['padded_batch_counts'],predicted_batches=r['full_service_prediction']['padded_batch_counts'],actual_useful_row_fraction=reference['useful_row_fraction'],predicted_useful_row_fraction=r['full_service_prediction']['useful_row_fraction'],timer_lateness_max=reference['nominal_host_release_lateness_max'],timer_lateness_mean=reference['nominal_host_release_lateness_mean'],delivery_delay_max=reference['max_delivery_seconds'],queue_mean=reference['mean_queue_seconds'],request_mean=reference['mean_request_seconds'],outside_request_range=[sum(not 1<=len(b['deliveries'])<=8 for b in s['batches']) for s in samples],outside_padded_row_range=[sum(not 4<=b['padded_rows']<=1024 for b in s['batches']) for s in samples])
        rows.append(row)
rankings=[]
for actors in (4,16,128):
    cases=[r for r in rows if r['actors']==actors and r['status']=='completed']
    if len(cases)!=3:
        rankings.append({'actors':actors,'status':'unavailable'});continue
    predicted=min(r['predicted_seconds'] for r in cases);winners=[r['max_batch'] for r in cases if abs(r['predicted_seconds']-predicted)<1e-9];actual=min(cases,key=lambda r:r['median_seconds'])['max_batch']
    rankings.append({'actors':actors,'predicted_best_batches':winners,'actual_best_batch':actual,'prediction_preserved':actual in winners})
summary={'run':allocation['run_id'],'source':allocation['source'],**json.loads((root/'verification.json').read_text()),'analysis_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),'frozen_model_sha256':hashlib.sha256(fit_path.read_bytes()).hexdigest(),'cells':rows,'rankings':rankings,'limits':['Prospective fixed-coefficient cross-rental check, with forward calibration from the current rental.','Real one-process serving with virtual host delays; not distributed collection, IPC, network or learning.','Input pool rows selected deterministically differ from original game inputs.','Declared repeat1 phase costs are worker wall intervals, not GPU kernel occupancy.','Request and padded-row extrapolation are counted explicitly; all outliers and hash differences remain.']}
(root/'summary.json').write_text(json.dumps(summary,indent=2)+'\n')
hardware=[]
forward={}
for size in (1,4,16,64,256,1024):
    name=f'w64-d1-model-b{size}-s0';path=evidence/name/'result.json'
    if by_name.get(name,{}).get('status')!='completed':
        continue
    r=json.loads(path.read_text())
    rates={phase:median(w['observations']/w['seconds'] for w in r[phase]['windows']) for phase in ('inference','optimizer')}
    forward[size]=size/rates['inference']
    hardware.append({'card':'h100','capacity':'w64-d1','cell':name,'kind':'model','status':'completed','batch':size,'inference_observations_per_second':rates['inference'],'optimizer_observations_per_second':rates['optimizer'],'peak_allocated_bytes':max(r[p]['peak_allocated_bytes'] for p in ('inference','optimizer')),'source_sha256':hashlib.sha256(path.read_bytes()).hexdigest(),'attention_slots':r['attention_slots'],'input_sha256':r['input_sha256']})
(root/'capacity-report.json').write_text(json.dumps({'run':summary['run'],'hardware':hardware},indent=2)+'\n')
if len(forward)==6:
    exported={**model,'forward_seconds':forward}
    (root/'service-model.json').write_text(json.dumps({'schema':'manabot.research.serving-cost.v1','model':exported,'overhead_source':str(fit_path),'overhead_source_sha256':summary['frozen_model_sha256'],'forward_run':summary['run'],'summary_sha256':hashlib.sha256((root/'summary.json').read_bytes()).hexdigest(),'limits':summary['limits']},indent=2)+'\n')
for r in rows:print(r['name'],r['status'],round(r.get('actual_over_predicted',0),3),r.get('service_phase_seconds'))
print(rankings)
destination=Path('scratch/gpu-probes/serving-007');staging=destination.with_name('.serving-007-staging');assert not destination.exists() and not staging.exists();staging.mkdir()
for p in evidence.rglob('*'):
    if p.is_file() and p.suffix in {'.json','.log'}:
        target=staging/'evidence'/p.relative_to(evidence);target.parent.mkdir(parents=True,exist_ok=True);shutil.copy2(p,target)
for name in ('summary.json','verification.json','capacity-report.json','service-model.json'):
    if (root/name).exists():shutil.copy2(root/name,staging/name)
for name in ('seventh-cleanup.json','rental-costs-007.json','source-copy.json','job-007.json','preflight-007.json','seventh-pod.json','prepared-source.json','budget.json'):
    shutil.copy2(Path('.runs/gpu-validation')/name,staging/name)
shutil.copy2(__file__,staging/'analysis.source.txt');shutil.copy2('scratch/serving-confirmation-protocol.txt',staging/'protocol.txt');shutil.copy2('.runs/gpu-validation/observed-007.log',staging/'observed.log');staging.rename(destination)
