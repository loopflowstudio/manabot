"""Summarize complete diagnostic spans and censored active phases without speed claims."""
from datetime import datetime
import hashlib
import json
from pathlib import Path
import shutil


def main() -> None:
    private=Path(__file__).parent
    root=private/'result-010';evidence=root/'evidence'
    ledger=json.loads((private/'budget.json').read_text())
    allocation=next(a for a in ledger['allocations'] if a['name']=='h100-validation-010')
    assert allocation['status']=='terminal-deletion-confirmed'
    attempts=json.loads((evidence/'attempts.json').read_text())
    prior_path=Path('scratch/gpu-probes/collector-009/evidence/collector-s4096-seed10350/result.json')
    prior=json.loads(prior_path.read_text())['samples'][0]['identity']
    rows=[];identities:dict[int,list[dict[str,str]]]={}
    for attempt in attempts:
        name=attempt['name']
        if not name.startswith('phase-'):continue
        path=evidence/name
        row={'name':name,'status':attempt['status'],'child_seconds':attempt['seconds']}
        if (path/'supervision.json').exists():
            supervision=json.loads((path/'supervision.json').read_text())
            row['supervision']={k:v for k,v in supervision.items() if k!='memory'}
            observed=[s['rss_bytes'] for s in supervision['memory'] if s['rss_bytes'] is not None]
            row['peak_sampled_rss_bytes']=max(observed) if observed else None
        events=[];incomplete_lines=0
        if (path/'events.jsonl').exists():
            for line in (path/'events.jsonl').read_text().splitlines():
                try:events.append(json.loads(line))
                except json.JSONDecodeError:incomplete_lines+=1
        active=[];spans=[]
        for event in events:
            if event['kind']=='begin':active.append(event)
            else:
                assert event['kind']=='end' and active
                first=active.pop();assert first['name']==event['name'] and first['stack']==event['stack']
                spans.append({'name':event['name'],'stack':event['stack'],'seconds':event['monotonic']-first['monotonic'],'user_cpu_seconds':event['user_cpu']-first['user_cpu'],'system_cpu_seconds':event['system_cpu']-first['system_cpu'],'minor_faults':event['minor_faults']-first['minor_faults'],'major_faults':event['major_faults']-first['major_faults'],'start_peak_rss_bytes':first['peak_rss_bytes'],'end_peak_rss_bytes':event['peak_rss_bytes']})
        row['complete_spans']=spans;row['active_spans_at_end']=active;row['incomplete_event_lines']=incomplete_lines
        windows=[]
        for outer in spans:
            if not (outer['name'].startswith('warm-') or outer['name'].startswith('measured-')):continue
            phases={}
            for phase in ('gather_upload','forward','native_step','build_batch','post_collect_sync'):
                matching=[s for s in spans if s['name']==phase and outer['name'] in s['stack']]
                phases[phase]={'calls':len(matching),'seconds':sum(s['seconds'] for s in matching),'cpu_seconds':sum(s['user_cpu_seconds']+s['system_cpu_seconds'] for s in matching),'minor_faults':sum(s['minor_faults'] for s in matching),'major_faults':sum(s['major_faults'] for s in matching)}
            remainder=outer['seconds']-sum(p['seconds'] for p in phases.values());assert remainder>=-1e-7
            windows.append({'name':outer['name'],'seconds':outer['seconds'],'phases':phases,'other_seconds':remainder})
        row['windows']=windows
        if (path/'result.json').exists():
            record=json.loads((path/'result.json').read_text());row['result']=record
            for sample in record['samples']:
                assert sample['learner_transitions']==32768 and sample['truncations']==0
                assert sample['exact_within_cell'] and sample['storage_independent']
                identities.setdefault(record['seed'],[]).append(sample['identity'])
            if record['seed']==10350:
                row['prior_009_identity_matches']=[s['identity']==prior for s in record['samples']]
        rows.append(row)
    assert len(rows)==4
    parity={str(seed):{'retained_samples':len(values),'all_exact':all(v==values[0] for v in values)} for seed,values in identities.items()}
    summary={'run':allocation['run_id'],'source':allocation['source'],**json.loads((root/'verification.json').read_text()),'analysis_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),'prior_009_sha256':hashlib.sha256(prior_path.read_bytes()).hexdigest(),'identity_comparisons':parity,'cells':rows,'limits':['Synchronized instrumented attribution; no throughput ratio or training-speed claim.','Process-isolated variants change history relative to run009 ABBA/BAAB.','A missing event end is censored, not a complete phase cost. Parent RSS samples may miss peaks.','Cgroup counters may include other processes; no single-component cause follows merely from correlation.']}
    (root/'summary.json').write_text(json.dumps(summary,indent=2)+'\n')
    dest=Path('scratch/gpu-probes/phase-010');temporary=dest.with_name('.phase-010-staging')
    assert not dest.exists() and not temporary.exists();temporary.mkdir()
    for p in evidence.rglob('*'):
        if p.is_file() and p.suffix in {'.json','.jsonl','.log'}:
            target=temporary/'evidence'/p.relative_to(evidence);target.parent.mkdir(parents=True,exist_ok=True);shutil.copy2(p,target)
    for name in ('summary.json','verification.json'):shutil.copy2(root/name,temporary/name)
    for name in ('tenth-cleanup.json','monitor-cleanup-010.json','rental-costs-010.json','source-copy.json','job-010.json','launch-010.json','preflight-010.json','tenth-pod.json','prepared-source.json','budget.json','lifecycle-010.json','download-recovery-010.json','observed-010.log'):
        shutil.copy2(private/name,temporary/name)
    for name,target in (('analyze-010.py','analysis.source.txt'),('recover-010.py','download.source.txt'),('launch-wait-010.py','monitor.source.txt')):
        shutil.copy2(private/name,temporary/target)
    shutil.copy2('scratch/collector-phase-protocol.txt',temporary/'protocol.txt')
    attempt=json.loads((private/'lifecycle-010.json').read_text())[0]
    cleanup=json.loads((private/'monitor-cleanup-010.json').read_text())
    lifecycle={'run':attempt['run'],'finished_at':attempt['finished_at'],**cleanup,'post_job_until_deletion_seconds':(datetime.fromisoformat(cleanup['deleted_at'])-datetime.fromisoformat(attempt['finished_at'])).total_seconds(),'whole_rental_estimate_usd':allocation['estimated_cost_usd'],'meaning':'Outer packaging success does not mean all diagnostic children passed.'}
    (temporary/'lifecycle-summary.json').write_text(json.dumps(lifecycle,indent=2)+'\n')
    temporary.rename(dest)
    for r in rows:
        print(r['name'],r['status'],r.get('supervision',{}).get('status'),'active',[e['name'] for e in r['active_spans_at_end']],[(w['name'],round(w['seconds'],4)) for w in r['windows']])
    print('identity comparisons',parity)


if __name__=='__main__':main()
